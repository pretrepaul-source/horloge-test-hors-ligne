    <script>

        function padZero(n) {

            return n < 10 ? '0' + n : n;

        }


        function mettreAJourHorloge() {

            var maintenant = new Date();

            var jours = ['Dimanche', 'Lundi', 'Mardi', 'Mercredi', 'Jeudi', 'Vendredi', 'Samedi'];

            var moisNoms = ['janvier', 'février', 'mars', 'avril', 'mai', 'juin', 'juillet', 'août', 'septembre', 'octobre', 'novembre', 'décembre'];


            var jourSemaine = jours[maintenant.getDay()];

            var heureNum = maintenant.getHours();

            var minuteNum = maintenant.getMinutes();


            var moment = "Matin";

            if (heureNum >= 12 && heureNum < 18) {

                moment = "Après-midi";

            } else if (heureNum >= 18 && heureNum < 22) {

                moment = "Soir";

            } else if (heureNum >= 22 || heureNum < 5) {

                moment = "Nuit";

            }


            var heures = padZero(heureNum);

            var minutes = padZero(minuteNum);

            var heureTexte = heures + ":" + minutes;


            var jourMois = maintenant.getDate();

            var mois = moisNoms[maintenant.getMonth()];

            var annee = maintenant.getFullYear();

            var dateComplete = jourMois + " " + mois + " " + annee;


            // Injection du texte

            document.getElementById('jour-semaine').innerHTML = jourSemaine;

            document.getElementById('moment').innerHTML = moment;

            document.getElementById('heure').innerHTML = heureTexte;

            document.getElementById('date-complete').innerHTML = dateComplete;


            // --- GESTION DES COULEURS SELON L'HEURE ---

            var bodyStyle = document.body.style;

            var jourStyle = document.getElementById('jour-semaine').style;

            var momentStyle = document.getElementById('moment').style;

            var heureStyle = document.getElementById('heure').style;

            var dateStyle = document.getElementById('date-complete').style;


            if (heureNum >= 6 && heureNum < 12) {

                // MATIN

                bodyStyle.backgroundColor = "#0b131e";

                jourStyle.color = "#ffffff";

                momentStyle.color = "#8ab4f8";

                heureStyle.color = "#ffffff";

                dateStyle.color = "#9aa0a6";

            } 

            else if (heureNum >= 12 && heureNum < 18) {

                // APRÈS-MIDI

                bodyStyle.backgroundColor = "#000000";

                jourStyle.color = "#ffffff";

                momentStyle.color = "#f28b82";

                heureStyle.color = "#ffffff";

                dateStyle.color = "#bdc1c6";

            } 

            else if (heureNum >= 18 && heureNum < 22) {

                // SOIR

                bodyStyle.backgroundColor = "#140a00";

                jourStyle.color = "#ffd59f";

                momentStyle.color = "#ffb74d";

                heureStyle.color = "#ffe0b2";

                dateStyle.color = "#b08d79";

            } 

            else {

                // NUIT (22h à 6h)

                bodyStyle.backgroundColor = "#050000";

                jourStyle.color = "#5c1d1d";

                momentStyle.color = "#822626";

                heureStyle.color = "#a83232";

                dateStyle.color = "#421515";

            }

        }


        mettreAJourHorloge();

        setInterval(mettreAJourHorloge, 1000);

    </script>
